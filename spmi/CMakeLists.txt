cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(pico_spmi_sniffer C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(pico_spmi_sniffer main.c)
target_link_libraries(pico_spmi_sniffer pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_enable_stdio_usb(pico_spmi_sniffer 1)
pico_enable_stdio_uart(pico_spmi_sniffer 0)
pico_add_extra_outputs(pico_spmi_sniffer)
